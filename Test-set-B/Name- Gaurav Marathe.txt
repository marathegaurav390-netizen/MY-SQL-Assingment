Q16)  SELECT 
    c.customer_id,
    c.customer_name,
    AVG(o.total_amount) AS avg_order_value
FROM Customers c
JOIN Orders o 
    ON c.customer_id = o.customer_id
GROUP BY 
    c.customer_id,
    c.customer_name
HAVING COUNT(o.order_id) > 2;
